import numpy as np
import matplotlib.pyplot as plt
from matplotlib.patches import Rectangle
import seaborn as sns

def plot_hp_box_swarm(ax, data, idx_spec=None, label_spec=None, null_dist=None, connection_map=None, clrs_spec=None, metric='sd'):
    if idx_spec is None:
        idx_spec = []
    if label_spec is None:
        label_spec = []
    if connection_map is None:
        connection_map = [1] * len(data)  # Default to connecting all groups
    if clrs_spec is None:
        clrs_spec = sns.color_palette('Set1', n_colors=len(idx_spec))
    
    # Plot null distributions
    if null_dist is not None:
        for i, group_null in enumerate(null_dist):
            if group_null:
                lower, upper = np.percentile(group_null, [5, 95])
                rect = Rectangle((i - 0.35, lower), 0.7, upper - lower, color='gray', alpha=0.2)
                ax.add_patch(rect)
    
    # Plot data points
    for i, group in enumerate(data):
        group = np.array(group)
        # Separate special points
        special_points = group[idx_spec] if idx_spec else []
        regular_points = np.delete(group, idx_spec) if idx_spec else group

        # Plot regular points
        jitter = np.random.uniform(-0.25, 0.25, size=len(regular_points))
        ax.scatter([i + 1 + j for j in jitter], regular_points, s=80, 
                   edgecolor='gray', facecolor='white', alpha=0.7, linewidth=1.5)
    
    # Plot special points
    for i, group in enumerate(data):
        if idx_spec:
            special_points = np.array(data[i])[idx_spec]
            for j, point in enumerate(special_points):
                if metric == 'sd':
                    ax.scatter(i + 1, point, s=120, 
                               edgecolor=clrs_spec[j], facecolor='white', linewidth=2.5, zorder=3)
                elif metric == 'ccgp':
                    ax.scatter(i + 1, point, s=120, 
                               edgecolor=clrs_spec[j], facecolor='gray', linewidth=2.5, zorder=3)
                elif metric == 'ps':
                    ax.scatter(i + 1, point, s=120, marker='^', 
                               edgecolor=clrs_spec[j], facecolor='gray', linewidth=2.0, zorder=3)
    
    # Draw connections
    if idx_spec and connection_map:
        for i in range(1, len(data)):
            if connection_map[i]:
                for j, idx in enumerate(idx_spec):
                    previous_point = np.array(data[i - 1])[idx]
                    current_point = np.array(data[i])[idx]
                    ax.plot([i, i + 1], [previous_point, current_point], 
                            color=clrs_spec[j], linewidth=2.5, zorder=2)
    
    # Add legend
    if label_spec and idx_spec:
        handles = [
            plt.Line2D([0], [0], marker='o', color='w', markerfacecolor='white', 
                       markeredgecolor=clrs_spec[i], markersize=10, label=label_spec[i])
            for i in range(len(idx_spec))
        ]
        ax.legend(handles=handles, loc='upper right', title="Special Points")
    
    # Final plot adjustments
    ax.set_xticks(range(1, len(data) + 1))
    ax.set_xlim(0.5, len(data) + 0.5)
    ax.set_xlabel('Groups')
    ax.set_ylabel('Values')

import numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.utils import shuffle
from sklearn.model_selection import cross_val_predict
from sklearn.metrics import accuracy_score
from sklearn.preprocessing import StandardScaler

def sd(avg_array, nr_iterations, nr_samples):
    """
    Perform shattering dimensionality analysis for a group of cells.

    Parameters:
    avg_array (list): List of neurons, with firing rates grouped hierarchically by trial type.
    nr_iterations (int): Number of bootstrap iterations to perform.
    nr_samples (int): Number of trials of each condition to sample.

    Returns:
    tuple: perf, boot, nr_cells
    perf (ndarray): Decoding performance for every dichotomy across iterations.
    boot (ndarray): Null distribution constructed using trial-label shuffle.
    nr_cells (int): Number of cells used in analysis.
    """
    PosSet, NegSet = define_sets()
    nr_cells = len(avg_array)
    nr_folds = 5

    perf = np.full((PosSet.shape[0], nr_iterations), np.nan)
    boot = np.full((PosSet.shape[0], nr_iterations), np.nan)

    for i in range(PosSet.shape[0]):
        gr_1 = PosSet[i]
        gr_2 = NegSet[i]

        for j in range(nr_iterations):
            training, testing = sample_from_data(avg_array, 0, nr_samples)
            training_data, _, labels_train, _ = prep_regressors(training, testing, gr_1, gr_2)

            # Normalize and remove NaN features
            scaler = StandardScaler()
            training_data = scaler.fit_transform(training_data)
            training_data = training_data[:, ~np.isnan(training_data).any(axis=0)]

            # Fit logistic regression model (equivalent to linear SVM)
            clf = LogisticRegression(max_iter=1000, solver='liblinear')
            y_hat = cross_val_predict(clf, training_data, labels_train, cv=nr_folds)
            perf[i, j] = accuracy_score(labels_train, y_hat)

            # Compute null distribution (shuffle labels)
            shuffled_labels = shuffle(labels_train)
            y_hat_null = cross_val_predict(clf, training_data, shuffled_labels, cv=nr_folds)
            boot[i, j] = accuracy_score(shuffled_labels, y_hat_null)

        print(f"Finished dichotomy {i + 1}")

    return perf, boot, nr_cells

def sample_from_data(array, test_samples, train_samples):
    """
    Sample data for training and testing.

    Returns:
    tuple: training, testing (both lists)
    """
    testing = [None] * len(array)
    training = [None] * len(array)

    for i in range(len(array)):
        testing[i] = []
        training[i] = []

        for trials in array[i]:
            test_indices = np.random.choice(len(trials), test_samples, replace=False)
            testing[i].append(trials[test_indices])

            remaining_trials = np.delete(trials, test_indices)
            train_indices = np.random.choice(len(remaining_trials), train_samples, replace=False)
            training[i].append(remaining_trials[train_indices])

    return training, testing

def prep_regressors(training, testing, gr_1, gr_2):
    """
    Prepare regressors for training and testing.

    Returns:
    tuple: training, testing, labels_train, labels_test
    """
    training_1 = np.vstack([np.vstack(training[i][gr_1]) for i in range(len(training))])
    training_2 = np.vstack([np.vstack(training[i][gr_2]) for i in range(len(training))])

    labels_train = np.concatenate([np.ones(len(training_1)), -1 * np.ones(len(training_2))])

    testing_1 = np.vstack([np.vstack(testing[i][gr_1]) for i in range(len(testing))])
    testing_2 = np.vstack([np.vstack(testing[i][gr_2]) for i in range(len(testing))])

    labels_test = np.concatenate([np.ones(len(testing_1)), -1 * np.ones(len(testing_2))])

    training = np.vstack([training_1, training_2])
    testing = np.vstack([testing_1, testing_2])

    return training, testing, labels_train, labels_test

import numpy as np

def make_groups(variables, var_names=None, binary=True, include_only=None):
    """
    Create groups with all possible combinations of given variables.

    Parameters:
    variables (ndarray): Array of variables, each column represents a variable.
    var_names (list of str): Names of the variables.
    binary (bool): Whether to create binary groups. Default is True.
    include_only (ndarray): Boolean mask to include only specific rows. Default is None.

    Returns:
    tuple:
        groups (ndarray): Array of groups representing combinations of variable levels.
        legend (list of str): Legend explaining each group's variable combination.
    """
    if var_names is None:
        var_names = [f'Var{i}' for i in range(variables.shape[1])]
    
    if include_only is None:
        include_only = np.ones(variables.shape[0], dtype=bool)

    # Get unique levels for each variable
    unique_levels = []
    for i in range(variables.shape[1]):
        unique_levels.append(np.unique(variables[include_only, i]))

    # Create all combinations of variable levels
    all_combinations = np.array(np.meshgrid(*unique_levels)).T.reshape(-1, len(unique_levels))

    # Create groups
    groups = []
    for combination in all_combinations:
        mask = np.all(variables[:, None, :] == combination, axis=2).any(axis=1)
        groups.append(mask.astype(int))
    groups = np.vstack(groups).T

    # Create legend
    legend = []
    for combination in all_combinations:
        legend.append(', '.join(f'{var_names[i]}={val}' for i, val in enumerate(combination)))

    return groups, legend
