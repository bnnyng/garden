def main():
    num_cores = os.cpu_count()
    print(f"Using {num_cores} CPU cores for processing.")

    comment_data = ALL_COMMENT_DATA.head(10)
    category_definitions = {'deductive': CATEGORY_DEFINITIONS['deductive']}
    tasks = []
    project_ids = list(comment_data['project-id'].unique())
    for project_id in project_ids[:1]:
        project_data = comment_data[comment_data['project-id'] == project_id]
        tasks.append({
            "name": project_id,
            "ids": project_data['id'].to_list(),
            "docs": project_data['content-cleaned'].to_list(),
        })
    with concurrent.futures.ProcessPoolExecutor(max_workers=num_cores) as executor:
        futures = [
            executor.submit(
                run_label_process, task["name"], 
                KEY, category_definitions,
                task["ids"], task["docs"], BATCH_SIZE
            ) for task in tasks
        ]
        for future in concurrent.futures.as_completed(futures):
            try:
                result = future.result()
                print("Process result:", result)
            except Exception as e:
                print(f"Error in process: {e}")
