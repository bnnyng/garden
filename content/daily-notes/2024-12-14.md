```python
def get_prop_correct_first_instance(beh):
    """
    Method for computing session-level accuracy on inference trials.

    :param beh: List of dictionaries, each containing behavioral data for one session.
    :return: List of NumPy arrays reporting session-level inference performance by block.
    """
    prop_correct = []
    
    for session_data in beh:
        # Extract task information
        context = session_data['task_info']['context']
        block_nr = get_block_nr(context)
        gt = session_data['task_info']['response_sequence']
        stim_seq = session_data['task_info']['stim_sequence']

        # Get the responses
        events_code = session_data['data']['events'][:, 1]
        response = np.where(np.isin(events_code, [31, 36]), events_code, np.nan)
        response[response == 36] = 0
        response[response == 31] = 1

        # Correct answers
        is_correct = (response == gt)

        # Get the instance number of the image in each block
        instance_nr = get_instance_nr(stim_seq, block_nr)

        # Identify the first instance in each block
        first_instance = instance_nr == 1
        block_first_instance = block_nr[first_instance]

        # Indexing the blocks
        unique_blocks = np.unique(block_first_instance)
        idx = [np.where(block_first_instance == block)[0] for block in unique_blocks]

        # Grab accuracy of all first instances
        is_correct_first = is_correct[first_instance]
        temp = [is_correct_first[block_idx] for block_idx in idx]
        temp = [np.pad(block, (0, 4 - len(block)), constant_values=np.nan) for block in temp]
        is_correct_first = np.vstack(temp)

        # Reshape to stack accuracy across blocks
        is_correct_first = is_correct_first.reshape(len(np.unique(stim_seq)), np.max(block_nr)).T

        # Append the last trial from the previous block
        which_trials = [np.where(block_nr == block)[0][-1] for block in unique_blocks]
        is_correct_last = is_correct[which_trials]

        # Combine the last trial and first instance accuracy
        session_result = np.column_stack((is_correct_last, is_correct_first))

        # Exclude the very first block
        session_result = session_result[1:, :]

        # Append the result for the current session
        prop_correct.append(session_result)

    return prop_correct
```

```python
import numpy as np

def split_sessions(beh):
    """
    Wrapper for computing significance of session-level inference behavior.

    Parameters:
        beh (dict): Dictionary containing behavioral data for all sessions.

    Returns:
        sess (dict): Dictionary with session grouping.
    """
    # Extract session names
    sessions = [x['sessionID'] for x in beh['data']]

    # Test inference trials
    perf = test_inference_trials(beh)

    # Group sessions based on significance
    absent = [sessions[i] for i in range(len(sessions))
              if perf['inference'][i] > 0.05 and perf['baseline'][i] < 0.05]
    present = [sessions[i] for i in range(len(sessions))
               if perf['inference'][i] < 0.05 and perf['baseline'][i] < 0.05]

    # Create output dictionary
    sess = {
        'absent': absent,
        'present': present,
        'absent_idx': [i for i, s in enumerate(sessions) if s in absent],
        'present_idx': [i for i, s in enumerate(sessions) if s in present],
        'sessions': sessions
    }

    return sess
```

```python
from scipy.stats import binom

def test_inference_trials(beh):
    """
    Computes session-level significance for inference and baseline trials.

    Parameters:
        beh (dict): Dictionary with behavioral data.

    Returns:
        stats (dict): Dictionary with significance and performance metrics.
    """
    # Session-level accuracy
    prop_correct = get_prop_correct_first_instance(beh)

    # Initialize outputs
    baseline = np.full(len(prop_correct), np.nan)
    inference = np.full(len(prop_correct), np.nan)
    inf_perf = np.full(len(prop_correct), np.nan)

    # Iterate over sessions
    for i, X in enumerate(prop_correct):
        # First inference trial is in column 3
        infr_trials = X[:, 2]
        valid_trials = np.sum(~np.isnan(infr_trials))

        # Significance testing
        baseline[i] = 1 - binom.cdf(np.sum(X[:, 0]), len(X[:, 0]), 0.5)
        inference[i] = 1 - binom.cdf(np.nansum(infr_trials), valid_trials, 0.5)

        # Proportion of correct inference trials
        inf_perf[i] = np.nansum(infr_trials) / valid_trials

    return {'baseline': baseline, 'inference': inference, 'inf_perf': inf_perf}
```
